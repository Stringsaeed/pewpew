export type LevelNumber =

export interface LevelSpec {
  readonly n: LevelNumber;
  /** Fixed per level, so players and the verifier fly the identical compiled track. */
  readonly seed: number;
  readonly length: number;        // world units to the wormhole (or boss arena)
  readonly scroll: number;        // units per second
  readonly density: number;       // hazards per 100 units
  readonly mix: { readonly asteroid: number; readonly mine: number; readonly drone: number }; // weights
  readonly drones: readonly DronePattern[];
  readonly droneFireRate: number; // shots per second per drone; 0 = drones hold fire
  readonly wells: number;         // count over the whole track
  readonly wellPull: number;      // lateral units/s^2 at the core
  readonly pickups: { readonly crystal: number; readonly shield: number; readonly rapid: number }; // per 100 units
  readonly boss: BossId | null;   // non-null exactly on 10, 20, 30 (verifier asserts)
}

export type Stars = 1 | 2 | 3;

export type Phase =
  | { readonly p: 'fly' }
  | { readonly p: 'boss' }

export interface Spawn { readonly y: number; readonly x: number; readonly kind: Kind; readonly variant: number; readonly param: number }
/** Compiled from (LevelSpec, seed) once at createWorld, sorted by y. */

export interface World {
  readonly spec: LevelSpec;
  readonly track: Track;
  cursor: number;           // next Spawn index
  tick: number;
  rng: number;              // mulberry32 state; the only randomness source

  score: number;

type Column = ArrayLike<number>; // readonly index signature: renderer and bots cannot write
type View<T> = T extends Float32Array | Uint8Array | Int16Array ? Column
  : T extends object ? { readonly [K in keyof T]: View<T[K]> } : T;
/** What the renderer, HUD, and bots get. Mutation does not typecheck. */
export type WorldView = View<World>;

export type Target = Kind | 'boss';
export type SimEvent =
  | { readonly e: 'start'; readonly level: LevelNumber }
  | { readonly e: 'fire'; readonly x: number; readonly y: number; readonly rapid: boolean }
  | { readonly e: 'enemyFire'; readonly x: number; readonly y: number }
  | { readonly e: 'hit'; readonly target: Target; readonly x: number; readonly y: number }     // shot landed, target survived
  | { readonly e: 'destroy'; readonly target: Target; readonly x: number; readonly y: number } // includes mine detonation
  | { readonly e: 'pickup'; readonly kind: PickupKind; readonly x: number; readonly y: number }
  | { readonly e: 'shieldBlock'; readonly x: number; readonly y: number }
  | { readonly e: 'playerHit'; readonly x: number; readonly y: number; readonly lives: number }

// ── src/render.ts: reads WorldView, owns GPU objects and particles. WGSL inline. Never imports React.

/** Kind → look. Exhaustive over Kind; 'glow' kinds draw as additive billboards, not meshes. */
export type Look =
  | { readonly draw: 'mesh'; readonly meshes: readonly MeshId[]; readonly tint: readonly [number, number, number] } // meshes[variant % len]
  | { readonly draw: 'glow'; readonly size: number; readonly rgba: readonly [number, number, number, number] };
export declare const LOOK: { readonly [K in Kind]: Look };

/**
 * Three pipelines, drawn in order into one pass with a depth buffer:
 *  stars  fullscreen triangle, procedural hashed star layers parallaxed by camera y uniform. No buffers.
 *  mesh   one shared vertex+index buffer holding every MeshData (interleaved pos|normal|color, 36 B), plus one
 *         instance buffer (x, y, rot, scale, tint rgb, flash = 8 f32). Per frame: counting-sort live bodies by MeshId,
 *         one writeBuffer, then drawIndexed(indexCount, n, firstIndex, baseVertex, firstInstance) per non-empty mesh.
 *         Rocket, boss, and wormhole are appended from player/boss/wormholeY scalars. Lambert + emissive vertex color.
 *  glow   instanced quads, additive blend, no depth write: bolts, exhaust, explosion particles, shield bubble.
 * Particles live here (CPU SoA, 2048 cap) and are spawned from SimEvents; the sim never knows they exist.
 */
export interface Renderer {
  /** world null = menu backdrop (stars only). alpha = leftover accumulator / STEP, for interpolation. */
  frame(world: WorldView | null, events: readonly SimEvent[], alpha: number, dtSec: number): void;
  dispose(): void;
}
/** Resizes the backing store from clientWidth × PixelRatio inside frame(); calls ctx.present() after submit. */
export function createRenderer(device: GPUDevice, ctx: GPUCanvasContext, format: GPUTextureFormat): Renderer { return NI(); }

// ── App.tsx: the shell. Owns screen (useReducer over flow.reduce), save, audio, renderer, the World, and the loop.
// Input → pixels is App.tsx (touch → Input, tick) → sim.ts (step) → render.ts (frame).

export interface Loop {
  world: World | null;  // replaced when (level, attempt) of the 'play' screen changes; null on menus
  input: Input;         // written by the touch responder, read once per step
  acc: number;
  last: number;
  readonly events: SimEvent[]; // reused buffer, cleared after renderer and audio consume it
}
/**
 * rAF body. stepping = screen.s === 'play'.
 *   acc += min(dt, 0.1); while (stepping && acc >= STEP) { step(world, input, events); acc -= STEP }
 *   renderer.frame(world, events, acc / STEP, dt); audio.events(events)
 *   for ev of events: won|lost → onEnded(ev)  (dispatch {a:'ended'}, writeSave(recordRun(...)))
 *   events.length = 0; setHud(h => hudOf(world, h))
 */
export function tick(l: Loop, now: number, stepping: boolean, r: Renderer, a: AudioEngine,
  onEnded: (ev: EventOf<'won'> | EventOf<'lost'>) => void): void { NI(); }

// ── scripts/verify-levels.ts: CI proof of "30 levels". Exit 1 on any failure; prints a per-level stats table.
//   assert LEVELS.length === 30, LEVELS[i].n === i + 1, (boss !== null) === (n % 10 === 0)
//   for each RAMP column marked 'nondecreasing': LEVELS[i][col] >= LEVELS[i-1][col]; boss hp 10 < 20 < 30
//   for n in 1..30: r = simulate(n, greedyBot); assert r.outcome === 'won' && r.dropped === 0
//                   assert replay(n, r.inputs).hash === r.hash   (determinism from the input stream alone)
//   difficulty sanity: idleBot loses every level >= 4; greedyBot's lives lost per tier is nondecreasing.
